// ==========================================
// Performance helpers: Cache + Lock
// เพิ่มเพื่อ (1) ลดการอ่านชีตซ้ำๆ ให้หน้าเว็บโหลดเร็วขึ้น
//           (2) กันข้อมูลชนกันเวลามีคนบันทึกพร้อมกันหลายคน
// ==========================================

// รอ lock ได้สูงสุดกี่ ms ก่อนจะตอบ error กลับไป (สั้นพอที่จะไม่ทำให้หน้าเว็บค้างนาน)
const LOCK_TIMEOUT_MS = 8000;

// อายุ cache (วินาที) ของแต่ละ action ที่อ่านข้อมูลซึ่งเปลี่ยนไม่บ่อย
const CACHE_TTL_SECONDS = {
  getInitData: 300,        // Config/Doctors/Staff แทบไม่เปลี่ยนระหว่างวัน
  getBeds: 20,              // เตียงเปลี่ยนบ่อยกว่า (admit/discharge) จึง cache สั้นๆ พอ
  getFocusTemplates: 600,
  getNursingTemplates: 600
};

// action การเขียนข้อมูลที่ต้องล็อกกันชนกัน (ตัดเฉพาะที่แก้ไขชีตที่ใช้ร่วมกันหลายคน)
const WRITE_ACTIONS_NEED_LOCK = new Set([
  'admitPatient', 'editPatient', 'addDoctor', 'moveBed', 'moveWard', 'discharge', 'movePatient',
  'saveAssessmentInitial', 'saveAssessmentPed',
  'saveClassification', 'saveClassificationSingle', 'deleteClassification',
  'saveClassificationPed', 'deleteClassificationPed',
  'saveDischargeDate',
  'saveFallRisk', 'saveFallRiskSingle', 'deleteFallRisk',
  'saveBradenScale', 'savePatientEdu',
  'saveFocusList', 'saveFocusTemplate',
  'saveNursingNotes', 'saveNursingTemplate',
  'saveDischargeRecord', 'saveNutritionAssessment',
  'dischargePatient', 'undoDischargePatient',
  'createServiceRequest', 'updateServiceRequestStatus'
]);

// เรียกฟังก์ชัน fn() โดยเช็ค cache ก่อน ถ้าไม่มีค่อยอ่านชีตจริงแล้วเก็บ cache ไว้
function cachedCall_(cacheKey, ttlSeconds, fn) {
  const cache = CacheService.getScriptCache();
  try {
    const cached = cache.get(cacheKey);
    if (cached !== null) return JSON.parse(cached);
  } catch (e) {
    // อ่าน cache ไม่ได้ก็ไปอ่านชีตตามปกติ ไม่ต้องทำให้ request พัง
  }

  const result = fn();

  try {
    const json = JSON.stringify(result);
    // CacheService จำกัดค่าไม่เกิน ~100KB ต่อ key ถ้าข้อมูลใหญ่เกินก็ข้ามการ cache ไปเฉยๆ
    if (json.length < 95000) cache.put(cacheKey, json, ttlSeconds);
  } catch (e) {
    // เก็บ cache ไม่ได้ก็ไม่เป็นไร ยังคืนผลลัพธ์ปกติ
  }

  return result;
}

// ล้าง cache key ที่ระบุ (ใช้หลังบันทึกข้อมูลสำเร็จ กันข้อมูลค้าง)
function clearCache_(keys) {
  if (!keys || !keys.length) return;
  try {
    CacheService.getScriptCache().removeAll(keys);
  } catch (e) {
    // ล้างไม่สำเร็จก็ปล่อยให้หมดอายุเองตาม TTL (สั้นอยู่แล้ว ไม่กระทบมาก)
  }
}

// หา cache key ของ action เขียนข้อมูล เพื่อล้างหลังบันทึกสำเร็จ
// ใช้ TTL สั้นสำหรับ 'beds' (20 วิ) เป็นตาข่ายนิรภัยอยู่แล้ว
// เคสไหนหา ward จาก payload ตรงๆ ไม่ได้ (เช่น dischargePatient ที่ดึง ward จากชีต) จะปล่อยให้ cache หมดอายุเอง
function getCacheKeysToInvalidate_(action, p) {
  switch (action) {
    case 'addDoctor':
      return ['initdata'];
    case 'saveFocusTemplate':
      return ['focustemplates'];
    case 'saveNursingTemplate':
      return ['nursingtemplates'];
    case 'admitPatient':
    case 'editPatient':
    case 'discharge': {
      const keys = [];
      if (p && p.ward) keys.push('beds_' + p.ward);
      return keys;
    }
    case 'moveBed':
    case 'moveWard':
    case 'movePatient': {
      const keys = [];
      if (p) {
        if (p.ward) keys.push('beds_' + p.ward);
        if (p.oldWard) keys.push('beds_' + p.oldWard);
        if (p.newWard) keys.push('beds_' + p.newWard);
        if (p.targetWard) keys.push('beds_' + p.targetWard);
      }
      return keys;
    }
    default:
      return [];
  }
}

function doGet(e) {
  const action = e.parameter.action;
  const ward = e.parameter.ward;
  const an = e.parameter.an;
  const sessionToken = e.parameter.sessionToken;

  // action ที่อ่านข้อมูลซึ่งเปลี่ยนไม่บ่อย -> อ่านผ่าน cache ก่อน ลดการอ่านชีตซ้ำ ทำให้โหลดเร็วขึ้น
  if (action === 'getInitData') return res(cachedCall_('initdata', CACHE_TTL_SECONDS.getInitData, function () { return getInitData(ward); }));
  if (action === 'getBeds') return res(cachedCall_('beds_' + ward, CACHE_TTL_SECONDS.getBeds, function () { return getBeds(ward); }));
  if (action === 'getFocusTemplates') return res(cachedCall_('focustemplates', CACHE_TTL_SECONDS.getFocusTemplates, function () { return getFocusTemplates(); }));
  if (action === 'getNursingTemplates') return res(cachedCall_('nursingtemplates', CACHE_TTL_SECONDS.getNursingTemplates, function () { return getNursingTemplates(); }));

  // action ที่เหลือ ข้อมูลเปลี่ยนบ่อยตามคนไข้แต่ละราย ไม่ cache เพื่อให้เห็นข้อมูลล่าสุดเสมอ
  if (action === 'getSessionUser') return res(getSessionUser(sessionToken));
  if (action === 'getPatients') return res(getPatients(ward));
  if (action === 'getAssessmentInitial') return res(getAssessmentInitial(an));
  if (action === 'getAssessmentPed') return res(getAssessmentPed(an));
  if (action === 'getClassifications') return res(getClassifications(an));
  if (action === 'getClassificationsPed') return res(getClassificationsPed(an));
  if (action === 'getFallRisk') return res(getFallRisk(an));
  if (action === 'getBradenScale') return res(getBradenScale(an));
  if (action === 'getPatientEdu') return res(getPatientEdu(an));
  if (action === 'getFocusList') return res(getFocusList(an));
  if (action === 'getNursingNotes') return res(getNursingNotes(an));
  if (action === 'getDischargeRecord') return res(getDischargeRecord(an));
  if (action === 'getNutritionAssessment') return res(getNutritionAssessment(an));
  return HtmlService.createHtmlOutput("API is running");
}

function doPost(e) {
  let data;
  try {
    data = JSON.parse(e.postData.contents);
  } catch (err) {
    return res({ status: 'error', message: 'Invalid JSON: ' + err.message });
  }

  // Route to CANDI chatbot directly if question is present
  if (data && data.question !== undefined) {
    return handleCandiChat_(data.question, data.context);
  }

  const action = data.action;
  const p = data.payload; // payload ที่ส่งมาจาก JavaScript

  let lock = null;
  let result;

  try {
    // ล็อกเฉพาะ action ที่เขียนข้อมูล เพื่อกันสองคนบันทึกชนกัน
    // (action ที่อ่านอย่างเดียว เช่น getServiceRequests จะไม่ถูกล็อก ไม่กระทบความเร็ว)
    if (WRITE_ACTIONS_NEED_LOCK.has(action)) {
      lock = LockService.getScriptLock();
      const gotLock = lock.tryLock(LOCK_TIMEOUT_MS);
      if (!gotLock) {
        return res({ status: 'error', message: 'ระบบกำลังบันทึกข้อมูลของผู้ใช้อื่นอยู่ กรุณาลองใหม่อีกครั้งในไม่กี่วินาที' });
      }
    }

    switch (action) {
      case 'admitPatient': result = admitPatient(p); break;
      case 'editPatient': result = editPatient(p); break;
      case 'addDoctor': result = addDoctor(p.name); break;
      case 'moveBed': result = moveBed(p); break;
      case 'moveWard': result = moveWard(p); break;
      case 'discharge': result = discharge(p); break;
      case 'movePatient': result = movePatient(p); break;
      case 'saveAssessmentInitial': result = saveAssessmentInitial(p.an, p.formData, p); break;
      case 'saveAssessmentPed': result = saveAssessmentPed(p.an, p.formData, p); break;

      case 'saveClassification': result = saveClassification(p); break;
      case 'saveClassificationSingle': result = saveClassificationSingle(data); break; // แก้ไขให้เป็น case
      case 'deleteClassification': result = deleteClassification(p); break;
      case 'saveClassificationPed': result = saveClassificationPed(p); break;
      case 'deleteClassificationPed': result = deleteClassificationPed(p); break;

      case 'saveDischargeDate': result = saveDischargeDate(p); break;

      case 'saveFallRisk': result = saveFallRisk(p); break;
      case 'saveFallRiskSingle': result = saveFallRiskSingle(data); break; // แก้ไขให้เป็น case
      case 'deleteFallRisk': result = deleteFallRisk(p); break;

      case 'saveBradenScale': result = saveBradenScale(p); break;
      case 'savePatientEdu': result = savePatientEdu(p); break;
      case 'saveFocusList': result = saveFocusList(p); break;
      case 'saveFocusTemplate': result = saveFocusTemplate(p); break;
      case 'saveNursingNotes': result = saveNursingNotes(p); break;
      case 'saveNursingTemplate': result = saveNursingTemplate(p); break;
      case 'saveDischargeRecord': result = saveDischargeRecord(p); break;
      case 'saveNutritionAssessment': result = saveNutritionAssessment(p); break;
      case 'dischargePatient': result = processDischargePatient(data); break;
      case 'searchDischargedPatients': result = searchDischargedPatients(p); break;
      case 'undoDischargePatient': result = undoDischargePatient(data); break;
      case 'sendRegistrationVerification': result = sendRegistrationVerification(p); break;
      case 'verifyRegistrationCode': result = verifyRegistrationCode(p); break;
      case 'loginUser': result = loginUser(p); break;
      case 'sendPasswordResetLink': result = sendPasswordResetLink(p); break;
      case 'resetPasswordWithToken': result = resetPasswordWithToken(p); break;
      case 'logoutUser': result = logoutUser(p); break;
      case 'getSessionUser': result = getSessionUser(p && p.sessionToken); break;
      case 'createServiceRequest': result = createServiceRequest(p); break;
      case 'getServiceRequests': result = getServiceRequests(p); break;
      case 'updateServiceRequestStatus': result = updateServiceRequestStatus(p); break;
      default: result = { status: 'error', message: 'Invalid Action' };
    }
  } finally {
    if (lock) lock.releaseLock();
  }

  // ล้าง cache เฉพาะตอนบันทึกสำเร็จ (status === 'success') เพื่อไม่ให้ข้อมูลเก่าค้างเกิน TTL
  if (result && result.status === 'success') {
    const keysToClear = getCacheKeysToInvalidate_(action, p);
    if (keysToClear.length) clearCache_(keysToClear);
  }

  return res(result);
}

}

// --- ฟังก์ชันดึงข้อมูลต่างๆ (คงเดิมตามระบบคุณ) ---
// ==========================================
// ฟังก์ชันจัดการวันที่ให้เป็น YYYY-MM-DD แบบปลอดภัย (แก้ปัญหาเวลาเหลื่อม 7 ชม.)
// ==========================================
function formatSafeDate(value) {
  if (!value) return "";
  // ตรวจสอบว่าเป็น Object Date แบบชัวร์ๆ
  if (Object.prototype.toString.call(value) === '[object Date]') {
    return Utilities.formatDate(value, "Asia/Bangkok", "yyyy-MM-dd");
  }
  // หากเป็น String ที่ติด Timezone มา (เช่น 2026-03-09T17:00:00.000Z)
  if (typeof value === 'string' && value.includes('T')) {
    return value.split('T')[0];
  }
  // กรณีเป็น String ปกติ หรืออื่นๆ
  return value;
}
function normalizeEvalDate(value) {
  if (!value) return "";
  if (Object.prototype.toString.call(value) === '[object Date]') {
    return Utilities.formatDate(value, APP_TIMEZONE, "yyyy-MM-dd");
  }

  const str = String(value).trim();
  if (!str) return "";

  const isoMatch = str.match(/(\d{4})-(\d{2})-(\d{2})/);
  if (isoMatch) return `${isoMatch[1]}-${isoMatch[2]}-${isoMatch[3]}`;

  const slashMatch = str.match(/(\d{1,2})\/(\d{1,2})\/(\d{4})/);
  if (slashMatch) {
    return `${slashMatch[3]}-${String(slashMatch[2]).padStart(2, '0')}-${String(slashMatch[1]).padStart(2, '0')}`;
  }

  if (str.includes('T')) {
    return str.split('T')[0];
  }

  const parsed = new Date(str);
  if (!isNaN(parsed.getTime())) {
    return Utilities.formatDate(parsed, APP_TIMEZONE, "yyyy-MM-dd");
  }

  return str;
}
function spreadsheetSerialToDate(value) {
  if (typeof value !== 'number' || !isFinite(value)) return null;
  const utcMs = Math.round((value - 25569) * 86400 * 1000);
  const date = new Date(utcMs);
  return isNaN(date.getTime()) ? null : date;
}
function normalizeSheetDisplayText(rawValue, displayValue) {
  const displayText = String(displayValue || '').trim();
  if (displayText) return displayText;
  if (rawValue === null || rawValue === undefined) return '';
  if (Object.prototype.toString.call(rawValue) === '[object Date]') {
    return Utilities.formatDate(rawValue, APP_TIMEZONE, 'yyyy-MM-dd HH:mm:ss');
  }
  return String(rawValue).trim();
}
function normalizeCurrentPatientCell(header, rawValue, displayValue) {
  const key = String(header || '').trim().toLowerCase();
  if (rawValue === null || rawValue === undefined) return '';

  const isDateField =
    key === 'date' ||
    key === 'dob' ||
    key === 'admitdate' ||
    key === 'dischargedate' ||
    key === 'วันที่' ||
    key === 'วันเกิด' ||
    key === 'วันที่จำหน่าย';

  if (isDateField) {
    return normalizeEvalDate(rawValue) || normalizeEvalDate(displayValue) || '';
  }

  const isTimeField =
    key === 'time' ||
    key === 'admittime' ||
    key === 'เวลา';

  if (isTimeField) {
    if (Object.prototype.toString.call(rawValue) === '[object Date]') {
      return Utilities.formatDate(rawValue, APP_TIMEZONE, 'HH:mm');
    }
    const rawText = String(rawValue || '').trim();
    if (!rawText) return '';
    if (rawText.includes('T')) {
      const timePart = rawText.split('T')[1] || '';
      return timePart.substring(0, 5);
    }
    const match = rawText.match(/\d{1,2}:\d{2}/);
    if (match) return match[0];
    return String(displayValue || rawText).trim();
  }

  return normalizeSheetDisplayText(rawValue, displayValue);
}
function normalizeRegistrySearchDate(rawValue, displayValue) {
  if (Object.prototype.toString.call(rawValue) === '[object Date]') {
    return Utilities.formatDate(rawValue, APP_TIMEZONE, 'yyyy-MM-dd');
  }
  if (typeof rawValue === 'number' && isFinite(rawValue)) {
    const parsedDate = spreadsheetSerialToDate(rawValue);
    if (parsedDate) {
      return Utilities.formatDate(parsedDate, APP_TIMEZONE, 'yyyy-MM-dd');
    }
  }

  const normalizedDisplay = normalizeEvalDate(displayValue);
  if (normalizedDisplay) return normalizedDisplay;

  return normalizeEvalDate(rawValue);
}
function isSameSheetIdentifier(rawValue, displayValue, targetValue) {
  const targetText = String(targetValue || '').trim();
  if (!targetText) return false;
  return normalizeSheetDisplayText(rawValue, displayValue) === targetText ||
    String(rawValue || '').trim() === targetText;
}
function normalizeHeaderKey(value) {
  return String(value || '').trim().toLowerCase();
}
function serializeRegistryHistoryValue(headerKey, rawValue, displayValue, dischargeDateValue) {
  if (headerKey === 'an' || headerKey === 'hn' || headerKey === 'ward' || headerKey === 'bed' || headerKey === 'ตึก' || headerKey === 'เตียง') {
    return normalizeSheetDisplayText(rawValue, displayValue);
  }

  if (headerKey === 'dischargedate' || headerKey === 'วันที่จำหน่าย') {
    return normalizeEvalDate(dischargeDateValue || rawValue || displayValue);
  }

  if (Object.prototype.toString.call(rawValue) === '[object Date]') {
    return Utilities.formatDate(rawValue, APP_TIMEZONE, 'yyyy-MM-dd');
  }

  return displayValue !== undefined && displayValue !== null && displayValue !== '' ? displayValue : rawValue;
}
function buildSerializedHistoryRow(sourceHeaders, rawRow, displayRow, targetHeaders, dischargeDateValue) {
  const sourceIndexMap = {};
  sourceHeaders.forEach(function(header, index) {
    sourceIndexMap[normalizeHeaderKey(header)] = index;
  });

  return targetHeaders.map(function(targetHeader) {
    const headerKey = normalizeHeaderKey(targetHeader);
    const sourceIndex = sourceIndexMap[headerKey];
    const rawValue = sourceIndex != null ? rawRow[sourceIndex] : '';
    const displayValue = sourceIndex != null ? displayRow[sourceIndex] : '';
    return serializeRegistryHistoryValue(headerKey, rawValue, displayValue, dischargeDateValue);
  });
}

function hashPassword(password, salt) {
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    String(salt || '') + String(password || ''),
    Utilities.Charset.UTF_8
  );
  return bytes.map(function(byte) {
    const normalized = byte < 0 ? byte + 256 : byte;
    return ('0' + normalized.toString(16)).slice(-2);
  }).join('');
}

function ensureSheetHeaders(sheet, headers) {
  const targetHeaders = headers.slice();
  const currentLastColumn = Math.max(sheet.getLastColumn(), 1);

  return sheet;

}

function parseDataUrl(dataUrl) {
  const match = String(dataUrl || '').match(/^data:([^;]+);base64,(.+)$/);
  if (!match) return null;
  return {
    mimeType: match[1],
    base64: match[2]
  };
}
function uploadServiceAttachments(requestId, attachments) {
  const normalizedAttachments = Array.isArray(attachments) ? attachments : [];
  if (normalizedAttachments.length === 0) return [];

  const folder = getOrCreateServiceUploadFolder();
  return normalizedAttachments.map(function(item, index) {
    const parsed = parseDataUrl(item.dataUrl);
    if (!parsed) {
      throw new Error('ไฟล์แนบมีรูปแบบไม่ถูกต้อง');
    }

    const rawBytes = Utilities.base64Decode(parsed.base64);
    const blob = Utilities.newBlob(rawBytes, parsed.mimeType, `${requestId}-${index + 1}-${String(item.name || 'attachment').trim() || 'attachment'}`);
    const file = folder.createFile(blob);
    return {
      name: String(item.name || file.getName()).trim(),
      url: file.getUrl(),
      fileId: file.getId(),
      mimeType: parsed.mimeType,
      size: Number(item.size || rawBytes.length || 0)
    };
  });
}

function normalizeShiftLabel(value) {
  const shift = String(value || '').trim();
  if (!shift) return '';
  if (shift.includes('ดึก') || shift.includes('ตึก')) return 'ดึก';
  if (shift.includes('เช้า') || shift.includes('เข้า')) return 'เช้า';
  if (shift.includes('บ่าย')) return 'บ่าย';
  return shift;
}
function getShiftPriority(value) {
  return SHIFT_ORDER[normalizeShiftLabel(value)] || 999;
}
function normalizeTimestampValue(value, fallbackDate) {
  if (Object.prototype.toString.call(value) === '[object Date]') {
    return value.getTime();
  }

  if (value) {
    const parsed = new Date(value);
    if (!isNaN(parsed.getTime())) return parsed.getTime();
  }

  const dateKey = normalizeEvalDate(fallbackDate || value);
  if (!dateKey) return 0;

  const [year, month, day] = dateKey.split('-').map(Number);
  return new Date(year, month - 1, day, 12, 0, 0).getTime();
}
function findLatestMatchingRow(data, predicate, timestampColIndex) {
  let foundRow = -1;
  let latestTimestamp = -1;

  for (let i = 1; i < data.length; i++) {
    if (!predicate(data[i])) continue;

    const rowTimestamp = normalizeTimestampValue(
      timestampColIndex > -1 ? data[i][timestampColIndex] : '',
      data[i][4]
    );

    if (foundRow === -1 || rowTimestamp > latestTimestamp || (rowTimestamp === latestTimestamp && i + 1 > foundRow)) {
      foundRow = i + 1;
      latestTimestamp = rowTimestamp;
    }
  }

  return foundRow;
}
function normalizeShiftRecords(records) {
  const latestBySlot = {};

  records.forEach(function(record, index) {
    const evalDate = normalizeEvalDate(record.evalDate || record.date);
    const shift = normalizeShiftLabel(record.shift);
    if (!evalDate || !shift || !SHIFT_ORDER[shift]) return;

    const timestampValue = normalizeTimestampValue(record.timestamp, evalDate);
    const key = `${evalDate}__${shift}`;
    const normalized = Object.assign({}, record, {
      evalDate: evalDate,
      shift: shift,
      _sortTimestamp: timestampValue,
      _sourceIndex: record._sourceIndex != null ? Number(record._sourceIndex) : index
    });

    const current = latestBySlot[key];
    if (!current ||
        timestampValue > current._sortTimestamp ||
        (timestampValue === current._sortTimestamp && normalized._sourceIndex > current._sourceIndex)) {
      latestBySlot[key] = normalized;
    }
  });

  return Object.keys(latestBySlot)
    .map(function(key) { return latestBySlot[key]; })
    .sort(function(a, b) {
      if (a.evalDate !== b.evalDate) return a.evalDate < b.evalDate ? -1 : 1;
      const shiftDiff = getShiftPriority(a.shift) - getShiftPriority(b.shift);
      if (shiftDiff !== 0) return shiftDiff;
      if (a._sortTimestamp !== b._sortTimestamp) return a._sortTimestamp - b._sortTimestamp;
      return a._sourceIndex - b._sourceIndex;
    })
    .map(function(record) {
      delete record._sortTimestamp;
      delete record._sourceIndex;
      return record;
    });
}
function parseJsonSafely(value, fallback) {
  if (!value) return fallback;
  if (typeof value === 'object') return value;
  try {
    return JSON.parse(value);
  } catch (error) {
    return fallback;
  }
}
function deleteMatchingRows(sheet, predicate) {
  const range = sheet.getDataRange();
  const data = range.getValues();
  const displayData = range.getDisplayValues();
  if (data.length <= 1) return 0;

  const rowsToDelete = [];
  for (let i = 1; i < data.length; i++) {
    if (predicate(data[i], i + 1)) {
      rowsToDelete.push(i + 1);
    }
  }

  rowsToDelete.sort(function(a, b) { return b - a; });
  rowsToDelete.forEach(function(rowNumber) {
    sheet.deleteRow(rowNumber);
  });

  return rowsToDelete.length;
}

function getLatestScoresMap() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Patient_Classification (Adult) -> map of AN -> Category / Score
  const classMap = {};
  const classSheet = ss.getSheetByName('Patient_Classification');
  if (classSheet) {
    const data = classSheet.getDataRange().getValues();
    if (data.length > 1) {
      const headers = data[0].map(h => String(h).toLowerCase().trim());
      const idxAN = headers.indexOf('an');
      const idxCategory = headers.indexOf('category');
      const idxTotal = headers.indexOf('total');
      if (idxAN !== -1) {
        for (let i = 1; i < data.length; i++) {
          const an = String(data[i][idxAN]).trim();
          if (an) {
            const category = idxCategory !== -1 ? String(data[i][idxCategory]).trim() : '';
            const total = idxTotal !== -1 ? String(data[i][idxTotal]).trim() : '';
            classMap[an] = { category, total };
          }
        }
      }
    }
  }

  // 2. Classifications_Pediatric (Child) -> map of AN -> ClassType / Score
  const classPedMap = {};
  const classPedSheet = ss.getSheetByName('Classifications_Pediatric');
  if (classPedSheet) {
    const data = classPedSheet.getDataRange().getValues();
    if (data.length > 1) {
      const headers = data[0].map(h => String(h).toLowerCase().trim());
      const idxAN = headers.indexOf('an');
      const idxClassType = headers.indexOf('classtype');
      const idxScore = headers.indexOf('score');
      if (idxAN !== -1) {
        for (let i = 1; i < data.length; i++) {
          const an = String(data[i][idxAN]).trim();
          if (an) {
            const classType = idxClassType !== -1 ? String(data[i][idxClassType]).trim() : '';
            const score = idxScore !== -1 ? String(data[i][idxScore]).trim() : '';
            classPedMap[an] = { classType, score };
          }
        }
      }
    }
  }

  // 3. Fall_Risk_Morse (Adult Morse score + MAAS score) -> map of AN -> latest MorseTotal & MAASScore
  const morseMap = {};
  const maasMap = {};
  const morseSheet = ss.getSheetByName('Fall_Risk_Morse');
  if (morseSheet) {
    const data = morseSheet.getDataRange().getValues();
    if (data.length > 1) {
      const headers = data[0].map(h => String(h).toLowerCase().trim());
      const idxAN = headers.indexOf('an');
      const idxMorse = headers.indexOf('morsetotal');
      const idxMaas = headers.indexOf('maasscore');
      if (idxAN !== -1) {
        for (let i = 1; i < data.length; i++) {
          const an = String(data[i][idxAN]).trim();
          if (an) {
            if (idxMorse !== -1) morseMap[an] = String(data[i][idxMorse]).trim();
            if (idxMaas !== -1) maasMap[an] = String(data[i][idxMaas]).trim();
          }
        }
      }
    }
  }

  // 4. Braden_Scale (Adult Braden score) -> map of AN -> TotalScore & EvalDate
  const bradenMap = {};
  const bradenDateMap = {};
  const bradenSheet = ss.getSheetByName('Braden_Scale');
  if (bradenSheet) {
    const data = bradenSheet.getDataRange().getValues();
    if (data.length > 1) {
      const headers = data[0].map(h => String(h).toLowerCase().trim());
      const idxAN = headers.indexOf('an');
      const idxTotalScore = headers.indexOf('totalscore');
      const idxEvalDate = headers.indexOf('evaldate');
      if (idxAN !== -1 && idxTotalScore !== -1) {
        for (let i = 1; i < data.length; i++) {
          const an = String(data[i][idxAN]).trim();
          if (an) {
            bradenMap[an] = String(data[i][idxTotalScore]).trim();
            if (idxEvalDate !== -1 && data[i][idxEvalDate]) {
              bradenDateMap[an] = String(data[i][idxEvalDate]).trim();
            }
          }
        }
      }
    }
  }

  return { classMap, classPedMap, morseMap, maasMap, bradenMap, bradenDateMap };
}

  const WARD_COLUMN_INDEX = 2; // CurrentPatients คอลัมน์ C = ward
  
  const scores = getLatestScoresMap();

  return rows
    .map((row, rowIndex) => ({ row: row, displayRow: displayData[rowIndex + 1] || [] }))
    .filter(item => {
      if (!targetWard) return false;
      return String(item.row[WARD_COLUMN_INDEX] || '').trim() === targetWard;
    })
    .map(item => {
      const row = item.row;
      const displayRow = item.displayRow;
      let obj = {};
      headers.forEach((header, i) => {
        obj[header.toString().trim()] = normalizeCurrentPatientCell(
          header,
          row[i],
          displayRow[i]
        ); 
      });
      obj.date = obj.date || normalizeCurrentPatientCell('date', row[0], displayRow[0]);
      obj.time = obj.time || normalizeCurrentPatientCell('time', row[1], displayRow[1]);
      obj.ward = obj.ward || normalizeCurrentPatientCell('ward', row[2], displayRow[2]);
      obj.bed = obj.bed || normalizeCurrentPatientCell('bed', row[3], displayRow[3]);

      obj.hn = obj.hn || normalizeCurrentPatientCell('hn', row[6], displayRow[6]);
      obj.an = obj.an || normalizeCurrentPatientCell('an', row[7], displayRow[7]);
      obj.name = obj.name || normalizeCurrentPatientCell('name', row[8], displayRow[8]);
      obj.address = obj.address || normalizeCurrentPatientCell('address', row[9], displayRow[9]);
      obj.dob = obj.dob || normalizeCurrentPatientCell('dob', row[10], displayRow[10]);
      obj.ageDisplay = obj.ageDisplay || normalizeCurrentPatientCell('ageDisplay', row[11], displayRow[11]);
      obj.dept = obj.dept || normalizeCurrentPatientCell('dept', row[12], displayRow[12]);
      obj.cc = obj.cc || normalizeCurrentPatientCell('cc', row[13], displayRow[13]);
      obj.pi = obj.pi || normalizeCurrentPatientCell('pi', row[14], displayRow[14]);
      obj.dx = obj.dx || normalizeCurrentPatientCell('dx', row[15], displayRow[15]);
      obj.doctor = obj.doctor || normalizeCurrentPatientCell('doctor', row[16], displayRow[16]);
      
      const anStr = String(obj.an).trim();
      obj.latestClass = scores.classMap[anStr] || null;
      obj.latestClassPed = scores.classPedMap[anStr] || null;
      obj.latestMorse = scores.morseMap[anStr] || '';

      obj.latestBraden = scores.bradenMap[anStr] || '';

      return obj;
    });
}

function getInitData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const configSheet = ss.getSheetByName('Config');
  const configData = configSheet ? configSheet.getDataRange().getValues() : [];
  const wards = configData.filter(r => r[0] === 'Ward').map(r => r[1]);
  const depts = configData.filter(r => r[0] === 'Dept').map(r => r[1]);
  
  const docSheet = ss.getSheetByName('Doctors');
  const doctors = docSheet ? docSheet.getDataRange().getValues().flat().filter(String) : [];
  
  const staffSheet = ss.getSheetByName('Staff');
  let nurses = [];
  if (staffSheet) {
    const staffData = staffSheet.getDataRange().getValues();
    // สมมติ แถวแรกคือ Header (Name, Position) ดึงข้อมูลตั้งแต่แถวที่ 2
    nurses = staffData.slice(1).map(row => {
      return { name: row[0], position: row[1] };
    });
  }
  
  return { wards, depts, doctors, nurses };
}

function admitPatient(p) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  ss.getSheetByName('CurrentPatients').appendRow([
    p.date, p.time, p.ward, p.bed, p.receivedFrom, p.referFrom,
    p.hn, p.an, p.name, p.address, p.dob, p.ageDisplay,
    p.dept, p.cc, p.pi, p.dx, p.doctor, 'Active'
  ]);
  updateBedStatus(p.ward, p.bed, 'ไม่ว่าง');
  return {status: 'success'};
}

function discharge(p) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const currentSheet = ss.getSheetByName('CurrentPatients');
  const historySheet = ss.getSheetByName('RegistryHistory');
  const data = currentSheet.getDataRange().getValues();
  
  for (let i = 1; i < data.length; i++) {
    if (data[i][7].toString() === p.an.toString()) {
      historySheet.appendRow(data[i]);
      currentSheet.deleteRow(i + 1);
      updateBedStatus(p.ward, p.bed, 'ว่าง');
      break;
    }
  }
  return {status: 'success'};
}

function updateBedStatus(ward, bedName, status) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Beds');
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0].toString() === ward.toString() && data[i][1].toString() === bedName.toString()) {
      sheet.getRange(i + 1, 3).setValue(status);
      break;
    }
  }
}

function editPatient(p) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('CurrentPatients');
  const data = sheet.getDataRange().getValues();
  let rowToUpdate = -1;
  for (let i = 1; i < data.length; i++) {
    if (data[i][7].toString() === p.an.toString()) {
      rowToUpdate = i + 1;
      break;
    }
  }
  if (rowToUpdate !== -1) {
    sheet.getRange(rowToUpdate, 1, 1, 17).setValues([[
      p.date, p.time, p.ward, p.bed, p.receivedFrom, p.referFrom,
      p.hn, p.an, p.name, p.address, p.dob, p.ageDisplay,
      p.dept, p.cc, p.pi, p.dx, p.doctor
    ]]);
    return {status: 'success'};
  }
  return {status: 'error', message: 'Not found'};
}

function movePatient(p) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('CurrentPatients');
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][7].toString() === p.an.toString()) {
      sheet.getRange(i + 1, 3).setValue(p.newWard);
      sheet.getRange(i + 1, 4).setValue(p.newBed);
      updateBedStatus(p.oldWard, p.oldBed, 'ว่าง');
      updateBedStatus(p.newWard, p.newBed, 'ไม่ว่าง');
      break;
    }
  }
  return {status: 'success'};
}

// ฟังก์ชันสำหรับดึงข้อมูลการประเมินแรกรับของ AN นั้นๆ
function getAssessmentInitial(an) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Assessment_Initial');
  if (!sheet) return null;
  
  const data = sheet.getDataRange().getValues();
  const anString = an.toString();
  
  // ค้นหาจากล่างขึ้นบน (เพื่อให้ได้ข้อมูลที่อัปเดตล่าสุด)
  for (let i = data.length - 1; i > 0; i--) {
    if (data[i][2] && data[i][2].toString() === anString) {
      const jsonStr = data[i][62]; 
      if (jsonStr) {
        try {
          return JSON.parse(jsonStr); // แปลง JSON กลับเป็น Object
        } catch(e) {
          return null;
        }
      }
    }
  }
  return null; // ถ้าไม่พบข้อมูล
}

function saveClassification(p) {
  const sheet = getOrCreateSheet('Patient_Classification', [
    'Timestamp', 'AN', 'HN', 'Ward', 'EvalDate', 'Shift',
    'Q1', 'Q2', 'Q3', 'Q4', 'Q5', 'Q6', 'Q7', 'Q8',
    'Total', 'Category', 'Assessor'
  ]);
  const data = sheet.getDataRange().getValues();
  const anString = p.an.toString();
  const targetDate = normalizeEvalDate(p.evalDate);
  const targetShift = normalizeShiftLabel(p.shift);

  const rowIndex = findLatestMatchingRow(data, function(row) {
    return String(row[1] || '').trim() === anString &&
      normalizeEvalDate(row[4]) === targetDate &&
      normalizeShiftLabel(row[5]) === targetShift;
  }, 0);
  
  const rowData = [
    new Date(), p.an, p.hn, p.ward, targetDate, targetShift,
    p.scores[0], p.scores[1], p.scores[2], p.scores[3],
    p.scores[4], p.scores[5], p.scores[6], p.scores[7],
    p.total, p.category, p.assessor
  ];
  
  if (rowIndex !== -1) {
    sheet.getRange(rowIndex, 1, 1, rowData.length).setValues([rowData]);
    return { status: 'success', message: 'อัปเดตข้อมูลเรียบร้อยแล้ว' };
  } else {
    sheet.appendRow(rowData);
    return { status: 'success', message: 'บันทึกข้อมูลใหม่เรียบร้อยแล้ว' };
  }
}

function deleteClassification(p) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Patient_Classification');
    if (!sheet) return { status: 'success', message: 'ไม่พบข้อมูลที่จะลบ', deletedCount: 0 };

    const targetAN = String(p.an || '').trim();
    const targetDate = normalizeEvalDate(p.evalDate);
    const targetShift = normalizeShiftLabel(p.shift);
    if (!targetAN || !targetDate || !targetShift) {
      return { status: 'error', message: 'ข้อมูลสำหรับลบไม่ครบถ้วน' };
    }

    const deletedCount = deleteMatchingRows(sheet, function(row) {
      return String(row[1] || '').trim() === targetAN &&
        normalizeEvalDate(row[4]) === targetDate &&
        normalizeShiftLabel(row[5]) === targetShift;
    });

    return {
      status: 'success',
      message: deletedCount > 0 ? 'ลบข้อมูลรายเวรเรียบร้อยแล้ว' : 'ไม่พบข้อมูลที่ต้องการลบ',
      deletedCount: deletedCount
    };
  } catch (error) {
    return { status: 'error', message: error.message };
  }
}

// ==========================================
function saveNursingNotes(p) {
  try {
    const detailSheet = getOrCreateNursingNotesDetailSheet();
    const noteItems = Array.isArray(p.noteData) ? p.noteData : [];
    const defaults = {
      an: String(p.an || ''),
      hn: String(p.hn || ''),
      ward: String(p.ward || '')
    };
    const normalizedNotes = noteItems.map(function(item) {
      return normalizeNursingNoteItem(item, defaults);
    });

    const data = detailSheet.getDataRange().getValues();
    const existingRowsById = {};
    const rowsToDelete = [];
    const incomingIds = {};

    for (let i = 0; i < normalizedNotes.length; i++) {
      incomingIds[normalizedNotes[i].id] = true;
    }

    for (let i = 1; i < data.length; i++) {
      if (String(data[i][1] || '') !== defaults.an) continue;
      const noteId = String(data[i][4] || '');
      if (noteId) existingRowsById[noteId] = i + 1;
      if (noteId && !incomingIds[noteId]) rowsToDelete.push(i + 1);
    }

    const rowsToAppend = [];
    normalizedNotes.forEach(function(note) {
      const rowData = [
        new Date(),
        note.an,
        note.hn,
        note.ward,
        note.id,
        note.date,
        note.shift,
        note.time,
        note.focus,
        note.s,
        note.o,
        note.i,
        note.e,
        note.eTime,
        note.nurse,
        note.pos,
        JSON.stringify(note)
      ];

      if (existingRowsById[note.id]) {
        detailSheet.getRange(existingRowsById[note.id], 1, 1, rowData.length).setValues([rowData]);
      } else {
        rowsToAppend.push(rowData);
      }
    });

    if (rowsToAppend.length > 0) {
      const startRow = detailSheet.getLastRow() + 1;
      detailSheet.getRange(startRow, 1, rowsToAppend.length, rowsToAppend[0].length).setValues(rowsToAppend);
    }

    rowsToDelete.sort(function(a, b) { return b - a; }).forEach(function(rowNumber) {
      detailSheet.deleteRow(rowNumber);
    });

    return {
      status: 'success',
      message: 'บันทึก Nursing Note สำเร็จ',
      savedCount: normalizedNotes.length
    };
  } catch (error) {
    return { status: 'error', message: error.message };
  }
}

function sendRegistrationVerification(p) {
  try {
    const username = String((p && p.username) || '').trim();
    const prefix = String((p && p.prefix) || '').trim();
    const firstName = String((p && p.firstName) || '').trim();
    const lastName = String((p && p.lastName) || '').trim();
    const password = String((p && p.password) || '');
    const confirmPassword = String((p && p.confirmPassword) || '');
    const email = normalizeEmail(p && p.email);
    const fullName = normalizeFullName([prefix, firstName, lastName].filter(String).join(' '));

    if (!username || !prefix || !firstName || !lastName || !email || !password || !confirmPassword) {
      return { status: 'error', message: 'กรุณากรอกข้อมูลลงทะเบียนให้ครบถ้วน' };
    }
    if (!/^[a-zA-Z0-9._-]{4,30}$/.test(username)) {
      return { status: 'error', message: 'ชื่อผู้ใช้ต้องมี 4-30 ตัว และใช้ได้เฉพาะ a-z, A-Z, 0-9, จุด, ขีดล่าง, ขีดกลาง' };
    }
    if (!isValidEmail(email)) {
      return { status: 'error', message: 'รูปแบบอีเมลไม่ถูกต้อง' };
    }
    if (password !== confirmPassword) {
      return { status: 'error', message: 'รหัสผ่านและยืนยันรหัสผ่านไม่ตรงกัน' };
    }
    if (password.length < 6) {
      return { status: 'error', message: 'รหัสผ่านต้องมีอย่างน้อย 6 ตัวอักษร' };
    }

    const staffMember = getStaffMemberByFullName(fullName);
    if (!staffMember) {
      return {
        status: 'error',
        message: 'ไม่พบข้อมูลในฐานข้อมูลบุคลากร กรุณาตรวจสอบข้อมูลให้ถูกต้อง หรือติดต่อผู้ดูแลระบบ'
      };
    }

    const existingUser = findUserAccountByEmail(email);
    if (existingUser && String(existingUser.values[6] || '').toLowerCase() === 'active') {
      return { status: 'error', message: 'อีเมลนี้ถูกลงทะเบียนใช้งานแล้ว' };
    }
    const existingUsername = findUserAccountByUsername(username);
    if (existingUsername && String(existingUsername.values[6] || '').toLowerCase() === 'active') {
      return { status: 'error', message: 'ชื่อผู้ใช้นี้ถูกใช้งานแล้ว กรุณาเลือกชื่อผู้ใช้อื่น' };
    }

    const verificationSheet = getOrCreateAuthVerificationSheet();
    const salt = generatePasswordSalt();
    const passwordHash = hashPassword(password, salt);
    const code = generateNumericCode(6);
    const createdAt = new Date();
    const expiresAt = new Date(createdAt.getTime() + (AUTH_VERIFY_MINUTES * 60 * 1000));

    verificationSheet.appendRow([
      createdAt,
      'REGISTER',
      email,
      staffMember.name,
      staffMember.position,
      passwordHash,
      salt,
      code,
      expiresAt,
      '',
      username
    ]);

    MailApp.sendEmail({
      to: email,
      subject: 'รหัสยืนยันการลงทะเบียน IPD Nurse System',
      htmlBody:
        '<div style="font-family:Sarabun,Arial,sans-serif;font-size:14px;line-height:1.7;">' +
        '<p>เรียน ' + staffMember.name + '</p>' +
        '<p>รหัสยืนยันสำหรับลงทะเบียนเข้าใช้งานระบบคือ</p>' +

function verifyRegistrationCode(p) {
  try {
    const email = normalizeEmail(p && p.email);
    const code = String((p && p.code) || '').trim();
    if (!email || !code) {
      return { status: 'error', message: 'กรุณากรอกอีเมลและรหัสยืนยัน' };
    }

    const verificationSheet = getOrCreateAuthVerificationSheet();
    const verificationData = verificationSheet.getDataRange().getValues();
    let matchedRowIndex = -1;
    let matchedRow = null;

    for (let i = verificationData.length - 1; i >= 1; i--) {
      if (String(verificationData[i][1] || '') !== 'REGISTER') continue;
      if (normalizeEmail(verificationData[i][2]) !== email) continue;
      if (String(verificationData[i][7] || '').trim() !== code) continue;
      if (verificationData[i][9]) continue;
      matchedRowIndex = i + 1;
      matchedRow = verificationData[i];
      break;
    }

    if (!matchedRow) {
      return { status: 'error', message: 'รหัสยืนยันไม่ถูกต้องหรือหมดอายุแล้ว' };
    }

    const expiresAt = matchedRow[8];
    if (!(expiresAt instanceof Date) || expiresAt.getTime() < new Date().getTime()) {
      return { status: 'error', message: 'รหัสยืนยันหมดอายุแล้ว กรุณาขอรหัสใหม่' };
    }

    const usersSheet = getOrCreateUserAccountsSheet();
    const existingUser = findUserAccountByEmail(email);
    const rowData = [
      existingUser ? existingUser.values[0] || new Date() : new Date(),
      matchedRow[3],
      matchedRow[4],
      email,
      matchedRow[5],
      matchedRow[6],
      'active',
      new Date(),
      '',
      String(matchedRow[10] || '').trim(),
      existingUser ? String(existingUser.values[10] || 'user').trim() || 'user' : 'user'
    ];

    if (existingUser) {
      usersSheet.getRange(existingUser.rowIndex, 1, 1, rowData.length).setValues([rowData]);
    } else {
      usersSheet.appendRow(rowData);
    }

    verificationSheet.getRange(matchedRowIndex, 10).setValue(new Date());

    return {
      status: 'success',
      message: 'ยืนยันอีเมลและสร้างบัญชีเรียบร้อยแล้ว',
      user: {
        fullName: String(matchedRow[3] || '').trim(),
        position: String(matchedRow[4] || '').trim(),
        email: email,
        username: String(matchedRow[10] || '').trim()
      }
    };
  } catch (error) {
    return { status: 'error', message: error.message };
  }
}

function resetPasswordWithToken(p) {
  try {
    const email = normalizeEmail(p && p.email);
    const token = String((p && p.token) || '').trim();
    const password = String((p && p.password) || '');
    const confirmPassword = String((p && p.confirmPassword) || '');

    if (!email || !token || !password || !confirmPassword) {
      return { status: 'error', message: 'กรุณากรอกข้อมูลตั้งรหัสผ่านใหม่ให้ครบถ้วน' };
    }
    if (password !== confirmPassword) {
      return { status: 'error', message: 'รหัสผ่านและยืนยันรหัสผ่านไม่ตรงกัน' };
    }
    if (password.length < 6) {
      return { status: 'error', message: 'รหัสผ่านต้องมีอย่างน้อย 6 ตัวอักษร' };
    }

    const resetSheet = getOrCreatePasswordResetSheet();
    const data = resetSheet.getDataRange().getValues();
    let matchedRowIndex = -1;
    let matchedRow = null;
    for (let i = data.length - 1; i >= 1; i--) {
      if (normalizeEmail(data[i][1]) !== email) continue;
      if (String(data[i][4] || '').trim() !== token) continue;
      if (data[i][6]) continue;
      matchedRowIndex = i + 1;
      matchedRow = data[i];
      break;
    }

    if (!matchedRow) {
      return { status: 'error', message: 'ลิงก์ตั้งรหัสผ่านใหม่ไม่ถูกต้องหรือถูกใช้งานไปแล้ว' };
    }
    if (!(matchedRow[5] instanceof Date) || matchedRow[5].getTime() < new Date().getTime()) {
      return { status: 'error', message: 'ลิงก์ตั้งรหัสผ่านใหม่หมดอายุแล้ว กรุณาขอใหม่อีกครั้ง' };
    }

    const userAccount = findUserAccountByEmail(email);
    if (!userAccount || String(userAccount.values[6] || '').toLowerCase() !== 'active') {
      return { status: 'error', message: 'ไม่พบบัญชีผู้ใช้ที่ต้องการรีเซ็ตรหัสผ่าน' };
    }

    const salt = generatePasswordSalt();
    const passwordHash = hashPassword(password, salt);
    const usersSheet = getOrCreateUserAccountsSheet();
    usersSheet.getRange(userAccount.rowIndex, 5).setValue(passwordHash);
    usersSheet.getRange(userAccount.rowIndex, 6).setValue(salt);
    resetSheet.getRange(matchedRowIndex, 7).setValue(new Date());

    return {
      status: 'success',
      message: 'ตั้งรหัสผ่านใหม่เรียบร้อยแล้ว สามารถเข้าสู่ระบบด้วยรหัสผ่านใหม่ได้ทันที'
    };
  } catch (error) {
    return { status: 'error', message: error.message };
  }
}

function loginUser(p) {
  try {
    const username = String((p && p.username) || '').trim();
    const password = String((p && p.password) || '');

      return { status: 'error', message: 'กรุณากรอกชื่อผู้ใช้และรหัสผ่าน' };
    }

    if (normalizeUsername(username) === normalizeUsername(ADMIN_USERNAME) && password === ADMIN_PASSWORD) {
      const adminUser = {
        fullName: 'IT Nurse Admin',
        position: 'System Administrator',
        email: 'admin11450@ipdnurse.local',
        username: ADMIN_USERNAME,
        role: 'admin'
      };
      const adminSession = createAuthSession(adminUser);
      return {

        sessionToken: adminSession.token,
        user: adminUser
      };
    }

    getOrCreateUserAccountsSheet().getRange(userAccount.rowIndex, 9).setValue(new Date());

    return {
      status: 'success',
      message: 'เข้าสู่ระบบสำเร็จ',
      sessionToken: session.token,
      user: buildUserResponseFromRow(userAccount.values)
    };
  } catch (error) {
    return { status: 'error', message: error.message };
  }
}

    const details = String((p && p.details) || '').trim();

      session.user.position,
      category,
      subject,

      'ใหม่',
      '',
      now,
      session.user.username,
      JSON.stringify(attachments),
      department

      sendLineServiceNotification_(createdRequest);
    } catch (notifyError) {

      status: 'success',
      message: 'ส่งคำขอเรียบร้อยแล้ว',

    };
  } catch (error) {
    return { status: 'error', message: error.message };
  }

function processDischargePatient(data) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const currentSheet = ss.getSheetByName('CurrentPatients');
    const historySheet = ss.getSheetByName('RegistryHistory');
    
    if (!currentSheet || !historySheet) {
      return { status: 'error', message: 'ไม่พบชีต CurrentPatients หรือ RegistryHistory' };
    }
    
    const currentRange = currentSheet.getDataRange();
    const sheetData = currentRange.getValues();
    const sheetDisplayData = currentRange.getDisplayValues();
    if (sheetData.length < 2) {
      return { status: 'error', message: 'ไม่มีรายชื่อผู้ป่วยในชีตปัจจุบัน' };
    }
    
    const headers = sheetData[0];
    const historyHeaders = historySheet.getLastRow() > 0
      ? historySheet.getRange(1, 1, 1, historySheet.getLastColumn()).getDisplayValues()[0]
      : headers;
    const anIndex = headers.findIndex(h => String(h).toLowerCase() === 'an');
    
    // หา Index ของ Ward และ Bed
    const wardIndex = headers.findIndex(h => String(h).toLowerCase() === 'ward' || String(h) === 'ตึก');
    const bedIndex = headers.findIndex(h => String(h).toLowerCase() === 'bed' || String(h) === 'เตียง');
    
    if (anIndex === -1) {
      return { status: 'error', message: 'ไม่พบคอลัมน์ AN ในชีต CurrentPatients' };
    }
    
    let targetRow = -1;
    let rowDataToMove = [];
    let displayRowToMove = [];
    let patientWard = "";
    let patientBed = "";
    
    for (let i = 1; i < sheetData.length; i++) {
      if (isSameSheetIdentifier(sheetData[i][anIndex], sheetDisplayData[i][anIndex], data.an)) {
        targetRow = i + 1; 
        rowDataToMove = sheetData[i];
        displayRowToMove = sheetDisplayData[i];
        
        // ดึงชื่อตึกและเตียงของผู้ป่วยคนนี้ออกมา
        if (wardIndex !== -1) patientWard = normalizeSheetDisplayText(sheetData[i][wardIndex], sheetDisplayData[i][wardIndex]);
        if (bedIndex !== -1) patientBed = normalizeSheetDisplayText(sheetData[i][bedIndex], sheetDisplayData[i][bedIndex]);
        break;
      }
    }
    
    if (targetRow !== -1) {
      const serializedDischargeDate = normalizeEvalDate(new Date());
      const historyRow = buildSerializedHistoryRow(headers, rowDataToMove, displayRowToMove, historyHeaders, serializedDischargeDate);
      const insertRow = historySheet.getLastRow() + 1;
      historySheet.getRange(insertRow, 1, 1, historyRow.length).setValues([historyRow]);

      const textFormatColumns = [];
      historyHeaders.forEach(function(header, index) {
        const key = normalizeHeaderKey(header);
        if (key === 'an' || key === 'hn' || key === 'dischargedate' || key === 'วันที่จำหน่าย') {
          textFormatColumns.push(index + 1);
        }
      });
      textFormatColumns.forEach(function(columnNumber) {
        historySheet.getRange(insertRow, columnNumber).setNumberFormat('@');
      });

      currentSheet.deleteRow(targetRow);     
      
      // ✅ เพิ่มคำสั่งอัปเดตสถานะเตียงเป็น 'ว่าง'
      if (patientWard && patientBed) {
        updateBedStatus(patientWard, patientBed, 'ว่าง');
      }
      
      return { status: 'success', message: 'จำหน่ายผู้ป่วยเรียบร้อย' };
    } else {
      return { status: 'error', message: 'ไม่พบผู้ป่วยที่ต้องการจำหน่ายในระบบ' };
    }
    
  } catch (error) {
    return { status: 'error', message: 'เซิร์ฟเวอร์ขัดข้อง: ' + error.message };
  }
}

// ค้นหาผู้ป่วยที่จำหน่ายแล้ว (รายวัน/รายเดือน)
// ==========================================
function searchDischargedPatients(p) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const historySheet = ss.getSheetByName('RegistryHistory');
    if (!historySheet) return { status: 'error', message: 'ไม่พบชีต RegistryHistory' };

    const range = historySheet.getDataRange();
    const data = range.getValues();
    const displayData = range.getDisplayValues();
    if (data.length < 2) return { status: 'success', data: [] };

    const headers = displayData[0].map(function(header) {
      return String(header || '').trim().toLowerCase();
    });
    const result = [];
    const wardIndex = headers.findIndex(function(h) { return h === 'ward' || h === 'ตึก'; });
    
    // หาคอลัมน์ วันที่จำหน่าย
    const dateIndex = headers.findIndex(function(h) {
      return h === 'dischargedate' || h === 'วันที่จำหน่าย';
    });

    for (let i = 1; i < data.length; i++) {
       const rawRow = data[i];
       const displayRow = displayData[i];
       let obj = {};
       headers.forEach(function(h, index) {
         obj[h] = normalizeSheetDisplayText(rawRow[index], displayRow[index]);
       });
       
       // กรอง Ward ก่อน
       if (p.ward && wardIndex !== -1) {
         const rowWard = normalizeSheetDisplayText(rawRow[wardIndex], displayRow[wardIndex]);
         if (rowWard !== String(p.ward).trim()) continue;
       }
       
       // กรองตามวันที่ / เดือนที่เลือก
       if (dateIndex !== -1 && (p.searchDate || p.searchMonth)) {
           const dateString = normalizeRegistrySearchDate(rawRow[dateIndex], displayRow[dateIndex]);
           if (!dateString) continue;

           if (p.searchType === 'date' && p.searchDate && dateString !== String(p.searchDate).trim()) continue;
           if (p.searchType === 'month' && p.searchMonth && !dateString.startsWith(String(p.searchMonth).trim())) continue;
       }
       
       obj.isDischarged = true; 
       result.push(obj);
    }
    return { status: 'success', data: result };
  } catch(e) {
    return { status: 'error', message: e.message };
  }
}

function undoDischargePatient(data) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const currentSheet = ss.getSheetByName('CurrentPatients');
    const historySheet = ss.getSheetByName('RegistryHistory');

    const historyRange = historySheet.getDataRange();
    const historyData = historyRange.getValues();
    const historyDisplayData = historyRange.getDisplayValues();
    const headers = historyData[0];
    const anIndex = headers.findIndex(h => String(h).toLowerCase() === 'an');
    
    // หา Index ของ Ward และ Bed
    const wardIndex = headers.findIndex(h => String(h).toLowerCase() === 'ward' || String(h) === 'ตึก');
    const bedIndex = headers.findIndex(h => String(h).toLowerCase() === 'bed' || String(h) === 'เตียง');

    let targetRow = -1;
    let rowDataToMove = [];
    let patientWard = "";
    let patientBed = "";

    // หาตัวผู้ป่วยในชีต History
    for (let i = 1; i < historyData.length; i++) {
      if (isSameSheetIdentifier(historyData[i][anIndex], historyDisplayData[i][anIndex], data.an)) {
        targetRow = i + 1; 
        rowDataToMove = historyData[i];
        
        // ดึงชื่อตึกและเตียงของผู้ป่วยคนนี้ออกมา
        if (wardIndex !== -1) patientWard = historyData[i][wardIndex];
        if (bedIndex !== -1) patientBed = historyData[i][bedIndex];
        break;
      }
    }

    if (targetRow !== -1) {
      currentSheet.appendRow(rowDataToMove); // เอาเข้าหน้าปัจจุบัน
      historySheet.deleteRow(targetRow);     // ลบออกจากประวัติ
      
      // ✅ เพิ่มคำสั่งเปลี่ยนสถานะเตียงกลับเป็น 'ไม่ว่าง'
      if (patientWard && patientBed) {
        updateBedStatus(patientWard, patientBed, 'ไม่ว่าง');
      }

      return { status: 'success', message: 'ยกเลิกจำหน่ายเรียบร้อย' };
    } else {
      return { status: 'error', message: 'ไม่พบผู้ป่วยในระบบประวัติ' };
    }
  } catch (error) {
    return { status: 'error', message: 'เซิร์ฟเวอร์ขัดข้อง: ' + error.message };
  }
}

function deleteClassificationPed(p) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Classifications_Pediatric');
    if (!sheet) return { status: 'success', message: 'ไม่พบข้อมูลที่จะลบ', deletedCount: 0 };

    const targetAN = String(p.an || '').trim();
    const targetDate = normalizeEvalDate(p.evalDate || p.date);
    const targetShift = normalizeShiftLabel(p.shift);
    if (!targetAN || !targetDate || !targetShift) {
      return { status: 'error', message: 'ข้อมูลสำหรับลบไม่ครบถ้วน' };
    }

    const data = sheet.getDataRange().getValues();
    const headers = data[0].map(function(h) { return String(h).toLowerCase().trim(); });
    const anIdx = headers.indexOf('an');
    const evalDateIdx = headers.indexOf('evaldate');
    const dateIdx = headers.indexOf('date');
    const shiftIdx = headers.indexOf('shift');

    const deletedCount = deleteMatchingRows(sheet, function(row) {
      const rowDate = normalizeEvalDate(evalDateIdx !== -1 ? row[evalDateIdx] : row[dateIdx]);
      return String(row[anIdx] || '').trim() === targetAN &&
        rowDate === targetDate &&
        normalizeShiftLabel(row[shiftIdx]) === targetShift;
    });

    return {
      status: 'success',
      message: deletedCount > 0 ? 'ลบข้อมูลรายเวรเรียบร้อยแล้ว' : 'ไม่พบข้อมูลที่ต้องการลบ',
      deletedCount: deletedCount
    };
  } catch (error) {
    return { status: 'error', message: error.message };
  }
}

      '  - O (Objective): ' + (context.noteO || 'ไม่มีข้อมูล') + '\n' +

    'พยาบาลผู้ช่วย AI ประจำ IPD Nurse Workbench ของโรงพยาบาลสมเด็จพระยุพราชสว่างแดนดิน\n\n' +

    'ข้อกำหนดด้านความปลอดภัย (Clinical Guardrail & Disclaimer):\n' +
    '- คุณต้องระบุข้อความสั้นๆ เสมอในส่วนท้ายหรือในเนื้อหาของคำตอบว่า:\n' +
    '  "น้อง CANDI เป็นระบบช่วยตัดสินใจเชิงคลินิก การตัดสินใจทางการพยาบาลและการบันทึกข้อมูลจริงขึ้นอยู่กับพยาบาลวิชาชีพผู้ดูแล"\n' +
    '- ห้ามเปิดเผยข้อมูลที่เป็นส่วนตัวของผู้ป่วยจริง หากตรวจพบชื่อจริงให้แนะนำให้กรองออก\n\n' +
    'หากพยาบาลร้องขอให้ "ร่าง Nursing Note (SOAP)" หรือ "ร่าง S-O-I-E" ให้คุณจัดรูปแบบคำตอบให้มีสัญลักษณ์ขึ้นต้นหัวข้อชัดเจนดังนี้ เพื่อให้ระบบป้อนข้อมูลอัตโนมัติบนหน้าเว็บบริหารจัดการได้ง่าย:\n' +

        ]
      }
    ],
    systemInstruction: {
      parts: [
        { text: systemInstruction }
      ]
    },
    generationConfig: {
      temperature: 0.2,
      maxOutputTokens: 1200
    }
  };

  const fetchOptions = {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify(requestPayload),
    muteHttpExceptions: true
  };

  try {
    const response = UrlFetchApp.fetch(url, fetchOptions);
    const responseCode = response.getResponseCode();
    const responseText = response.getContentText();
    
    if (responseCode !== 200) {
      return res({
        status: 'error',
        message: 'CANDI กำลังอยู่ในช่วงพัฒนาค่ะ (ติดต่อกับ Gemini API ไม่สำเร็จ HTTP ' + responseCode + ')'

    }
    
    const responseJson = JSON.parse(responseText);

        responseJson.candidates[0] && 
        responseJson.candidates[0].content && 
        responseJson.candidates[0].content.parts && 
        responseJson.candidates[0].content.parts[0]) {

        status: 'error',
        message: 'ไม่ได้รับข้อความตอบกลับจาก AI โมเดล โปรดตรวจสอบความถูกต้องของ API Key หรือโควต้าการใช้งาน'
      });
    }
    
    return res({
      status: 'success',
      reply: replyText
    });

  } catch (error) {
    return res({
      status: 'error',
      message: 'CANDI กำลังอยู่ในช่วงพัฒนาค่ะ (เกิดข้อผิดพลาดทางเทคนิคของเซิร์ฟเวอร์: ' + error.message + ')'
    });
  }
}
